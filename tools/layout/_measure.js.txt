window.confirm = () => true; window.alert = () => {};
(function(){
 const out=[]; const say=(k,v)=>out.push(k+": "+v);
 setTimeout(()=>{
  try{
   const R=(el)=>el?Math.round(el.getBoundingClientRect().height):0;
   say("cssViewport", document.documentElement.clientWidth+"x"+document.documentElement.clientHeight);
   startGame({ side:"west", scenarioId:"cobra" });
   document.querySelectorAll("#newOperationModal,#storageNotice,#coach,#coachSpot,#resultScreen,.setup-modal,.modal,#operationSetup").forEach(el=>{el.hidden=true;el.style.display="none";});
   render();
   const card=document.querySelector(".unit-card"), stage=document.querySelector("#mapStage");
   const wrap=document.querySelector(".battlefield-wrap"), hint=document.getElementById("actionHint");
   const acts=document.querySelector(".hud-actions");
   say("emptyCardDisplay", card?getComputedStyle(card).display:"none");
   const pick=state.units.find(u=>u.owner==="player"&&u.type!=="battalionHQ");
   if(pick){ state.selectedId=pick.id; render(); }
   const rows=()=>{
     const sp=[...document.querySelectorAll(".unit-stats > span")].filter(s=>s.offsetParent!==null);
     return new Set(sp.map(s=>s.offsetTop)).size;
   };
   const report=(tag)=>{
     say(tag+"StatRows", rows());
     say(tag+"CardH", R(card)+" (내용 "+(card?card.scrollHeight:0)+")");
     say(tag+"StageH", stage?stage.clientHeight:0);
     say(tag+"TileW", document.querySelector(".tile")?.offsetWidth||0);
     say(tag+"MapRows", stage&&document.querySelector(".tile")?
        (stage.clientHeight/document.querySelector(".tile").offsetWidth).toFixed(1):"-");
     say(tag+"HintH", R(hint));
     say(tag+"HintBottom", Math.round(hint.getBoundingClientRect().bottom)
        +" / 창 "+window.innerHeight+" / 버튼띠 위 "+(acts?Math.round(acts.getBoundingClientRect().top):"-"));
   };
   report("보통");
   [...document.querySelectorAll(".unit-stats > span.key")].forEach(s=>{
     if(s.textContent.startsWith("행동")) s.querySelector("strong").textContent="편성 중 / 내일부터";
     if(s.textContent.startsWith("보급")) s.querySelector("strong").textContent="보급선 이탈";
   });
   const al=document.querySelector(".hud-alerts");
   if(al){ al.hidden=false; al.style.display="flex"; al.textContent="보급 두절 경보 · 사령부 고립 경보"; }
   fitBoardToScreen();
   report("최악");

     const av=document.querySelector(".unit-card-visual");
     const cs=getComputedStyle(av||card,"::after");
     card.scrollTop = 0;
     const cr=card.getBoundingClientRect();
     say("자세히표시", (av?"이름줄":"카드")+" "+cs.content+" / 카드바닥 "+Math.round(cr.bottom)+" / 스크롤여유 "+(card.scrollHeight-card.clientHeight));
   }
   if(card){
     const cb=card.getBoundingClientRect();
     const vis=[...document.querySelectorAll(".unit-stats > span")].filter(x=>x.offsetParent!==null)
       .map(x=>{const r=x.getBoundingClientRect();
         const inCard = r.top>=cb.top && r.bottom<=cb.bottom;
         const hit = document.elementFromPoint(Math.round(r.left+4), Math.round((r.top+r.bottom)/2));
         const ontop = hit && (hit===x || x.contains(hit));
         return (inCard&&ontop?"O ":(inCard?"가림 ":"X "))+x.textContent.trim();});

       return (c.className||c.id)+"="+Math.round(r.height)+"@"+Math.round(r.top);});

     say("상황판CSS", "pad "+getComputedStyle(hud).padding+" gap "+getComputedStyle(hud).gap+" 배치 "+getComputedStyle(hud).display+" "+getComputedStyle(hud).gridTemplateColumns);
     const st=hud.querySelector(".hud-stat");
     say("칸CSS", "h "+Math.round(st.getBoundingClientRect().height)+" pad "+getComputedStyle(st).padding+" 배치 "+getComputedStyle(st).display+" "+getComputedStyle(st).flexDirection);

     say("상황판경보", Math.round(hud.getBoundingClientRect().height));
   }

   {
     // 좌표가 있는 아군 목표. holdTurns 를 안 적은 목표도 기본 유지 턴이 붙어
     // 카드에는 「유지 0/n턴」이 그려지므로, 여기서 거르면 오히려 실제와 멀어진다.

       say("목표칸", o.label+" ("+o.x+", "+o.y+")");
       say("목표칸카드", Math.round(cb.height)+" / 내용 "+c.scrollHeight+" / 넘침 "+(c.scrollHeight-c.clientHeight));

         .map(x=>{const r=x.getBoundingClientRect();
           const inCard = r.top>=cb.top && r.bottom<=cb.bottom;
           const hit = document.elementFromPoint(Math.round(r.left+4), Math.round((r.top+r.bottom)/2));
           const ontop = hit && (hit===x || x.contains(hit));
           return (inCard&&ontop?"O ":(inCard?"가림 ":"X "))+x.textContent.trim();});
       say("목표칸항목", vis.join(" | "));
     } else say("목표칸", "없음");
   }

  }catch(e){ out.push("ERROR: "+e.message); }
  const T=out.join(String.fromCharCode(10));

 },1200);
})();
</script>
